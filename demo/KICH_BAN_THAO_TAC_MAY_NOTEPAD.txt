--- [BƯỚC 2.3: Đổi vai sang Quản lý (Manager) để đối chiếu toàn quyền giám sát] ---
👉 Copy dán vào SQL EDITOR:

RESET ROLE;
SET ROLE nv_manager;

SELECT 
    i.cashier_id, 
    e.full_name AS ten_nhan_vien, 
    e.username AS tai_khoan, 
    e.phone AS so_dien_thoai,
    e.emp_role AS vai_tro, 
    e.default_shift AS ca_lam_viec, 
    count(i.invoice_id) AS so_luong_hoa_don,
    sum(i.total_amount) AS tong_doanh_thu
FROM billing.invoice i
JOIN hr.employee e ON i.cashier_id = e.employee_id
GROUP BY i.cashier_id, e.full_name, e.username, e.phone, e.emp_role, e.default_shift
ORDER BY i.cashier_id;

RESET ROLE;

--- [BƯỚC 3.1: Đảm bảo quyền Owner & Xóa index cũ để đưa bảng về trạng thái nguyên bản] ---
👉 Copy dán vào SQL EDITOR:

RESET ROLE;
DROP INDEX IF EXISTS billing.idx_invoice_total_amount;


--- [BƯỚC 3.2: Đo khi CHƯA có Index -> Kết quả: Seq Scan (Quét tuần tự toàn bảng)] ---
👉 Copy dán vào SQL EDITOR (Chỉ chuột vào dòng chữ: 'Seq Scan on invoice'):

EXPLAIN (ANALYZE, BUFFERS)
SELECT invoice_id, customer_id, cashier_id, total_amount, status 
FROM billing.invoice 
WHERE total_amount = 630000;


--- [BƯỚC 3.3: Tạo B-Tree Index và đo lại -> Kết quả: Index Scan (Nhảy thẳng đến hóa đơn)] ---
👉 Copy dán vào SQL EDITOR (Chỉ chuột vào dòng chữ: 'Index Scan using idx_invoice_total_amount'):

-- 1. Tạo chỉ mục B-Tree trên cột số tiền hóa đơn:
CREATE INDEX idx_invoice_total_amount ON billing.invoice (total_amount);

-- 2. Bật ép dùng Index và chạy lại câu truy vấn:
SET enable_seqscan = off;

EXPLAIN (ANALYZE, BUFFERS)
SELECT invoice_id, customer_id, cashier_id, total_amount, status 
FROM billing.invoice 
WHERE total_amount = 630000;

SET enable_seqscan = on;

--- [BƯỚC 4.1: Bóc tách dữ liệu bán cấu trúc NoSQL từ cột JSONB (->> và @>)] ---
👉 Copy dán vào SQL EDITOR:

SELECT 
    computer_code, 
    computer_name,
    specs ->> 'cpu'     AS cpu_chip,
    specs ->> 'gpu'     AS card_man_hinh,
    specs ->> 'ram'     AS dung_luong_ram
FROM core.computer
WHERE specs ->> 'gpu' ILIKE '%RTX%';

-- Truy vấn kết hợp toán tử chứa tập con JSON Containment (@>):
EXPLAIN (ANALYZE, BUFFERS)
SELECT computer_code, computer_name, specs 
FROM core.computer 
WHERE specs @> '{"gpu": "RTX 4070"}'::jsonb;
