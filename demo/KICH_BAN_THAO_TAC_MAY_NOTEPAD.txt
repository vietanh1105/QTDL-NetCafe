Bước 1.3: Thử nghiệm rủi ro Fail-Safe (Chặn lỗi ràng buộc)

-- [ NEON SQL EDITOR / DBEAVER ] --
-- 1. Chạy BEGIN và UPDATE; mong đợi lỗi 23514, ck_customer_balance.
BEGIN;
UPDATE core.customer SET balance = -50000
WHERE customer_id = (SELECT min(customer_id) FROM core.customer);
-- Sau khi đọc lỗi, chạy riêng:
ROLLBACK;

-- 2. Phòng phải đang có máy tham chiếu; mong đợi lỗi RESTRICT.
BEGIN;
DELETE FROM core.room
WHERE room_id = (SELECT min(room_id) FROM core.computer);
-- Sau khi đọc lỗi, chạy riêng:
ROLLBACK;

Bước 2.1 & 2.2: Đóng vai Thu ngân Thảo và Chứng minh RLS

-- [ NEON SQL EDITOR / DBEAVER ] --
-- Dùng cùng một kết nối; tài khoản demo phải có quyền SET ROLE.
BEGIN;
SET LOCAL ROLE nv_thao;
SELECT current_user, hr.fn_get_current_employee_id() AS employee_id;
SELECT * FROM billing.invoice ORDER BY invoice_id;
ROLLBACK;

-- Đổi sang quản lý để đối chiếu:
BEGIN;
SET LOCAL ROLE nv_manager;
SELECT i.cashier_id, e.full_name AS ten_nhan_vien,
       count(*) AS so_luong_hd, sum(i.total_amount) AS tong_doanh_thu
FROM billing.invoice i JOIN hr.employee e ON i.cashier_id = e.employee_id
GROUP BY i.cashier_id, e.full_name ORDER BY i.cashier_id;
ROLLBACK;

Bước 3.1: Chạy truy vấn tìm hóa đơn theo số tiền khi chưa có Index (Seq Scan)

-- [ NEON SQL EDITOR / DBEAVER ] --
RESET ROLE;
BEGIN;
DROP INDEX IF EXISTS billing.idx_invoice_total_amount;
ANALYZE billing.invoice;
EXPLAIN (ANALYZE, BUFFERS)
SELECT invoice_id, customer_id, cashier_id, total_amount, status
FROM billing.invoice WHERE total_amount = 630000;
-- Giữ kết nối và chuyển sang Bước 3.2.

======================================================================
Bước 3.2: Tạo B-Tree Index trên số tiền và Đối chiếu lại (Index Scan)

-- [ NEON SQL EDITOR / DBEAVER ] --
-- Tiếp tục transaction của Bước 3.1 trong cùng kết nối.
CREATE INDEX idx_invoice_total_amount ON billing.invoice (total_amount);
ANALYZE billing.invoice;
EXPLAIN (ANALYZE, BUFFERS)
SELECT invoice_id, customer_id, cashier_id, total_amount, status
FROM billing.invoice WHERE total_amount = 630000;
ROLLBACK;
